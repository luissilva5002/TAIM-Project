Flutter Companion App Integration

Stream sensor data over local WebSocket directly into this Three.js engine

This web engine exposes a WebSocket listener. Run your Flutter app on mobile connected to the same Wi-Fi, then stream real-time pitch, yaw, depth slider, and teleport action commands using JSON payloads: 

// Flutter Companion App - WebSockets + Sensors snippet
import 'package:web_socket_channel/io.dart';
import 'package:sensors_plus/sensors_plus.dart';

final channel = IOWebSocketChannel.connect('ws://YOUR_PC_LOCAL_IP:8080');

// Stream Gyroscope / Orientation updates (Pitch & Yaw)
gyroscopeEvents.listen((GyroscopeEvent event) {
  channel.sink.add(jsonEncode({
    'type': 'AIM',
    'pitch': event.x * 0.05,
    'yaw': event.y * 0.05,
  }));
});

// Send Teleport Action Command
void triggerTeleport() {
  channel.sink.add(jsonEncode({
    'type': 'TELEPORT',
    'mode': isAirMode ? 'AIR' : 'GROUND',
    'depth': currentSliderDepth,
  }));
}
